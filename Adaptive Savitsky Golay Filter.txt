clc; clear all; close all

data = readmatrix('multi_line_data2.csv');
if isempty(data)
    error('File "multi_line_data2.csv" not found or empty.');
end

f = data(:,1);
mag = data(:,2);
N = length(mag);

if mod(N,2)==0
    N = N-1; 
    f = f(1:N); 
    mag = mag(1:N); 
end % ensure odd length
fprintf('Data points: %d\n', N);

win_min = 5; win_max = 51; win_step = 2;
order_min = 2; order_max = 4;
W_candidates = win_min:win_step:win_max;
P_candidates = order_min:order_max;

if N > 2000
    use_subset = true;
    idx_subset = round(linspace(1, N, 2000));
    f_sub = f(idx_subset);
    mag_sub = mag(idx_subset);
else
    use_subset = false;
    f_sub = f; 
    mag_sub = mag;
end

tic;
[mag_cv_sub, windows_cv_sub, orders_cv_sub] = adaptive_sg_cv(f_sub, mag_sub, W_candidates, P_candidates);
time_cv = toc;
fprintf('CV Optimization done in %3.3f s\n', time_cv);

if use_subset
    mag_cv = interp1(f_sub, mag_cv_sub, f, 'pchip', 'extrap');
    windows_cv = interp1(f_sub, windows_cv_sub, f, 'nearest', 'extrap');
    orders_cv = interp1(f_sub, orders_cv_sub, f, 'nearest', 'extrap');
else
    mag_cv = mag_cv_sub; 
    windows_cv = windows_cv_sub; 
    orders_cv = orders_cv_sub;
end

figure('Position', [50 50 1200 600]);
subplot(2,1,1);
plot(f, mag, 'b-', f, mag_cv, 'r-', 'LineWidth', 1.2);
title('CV-Optimized Savitzky-Golay'); 
xlabel('Freq'); ylabel('Mag');
legend('Original', 'CV-Optimized', 'Location', 'best'); grid on;

subplot(2,1,2);
plot(f, windows_cv, 'b-'); hold on;
plot(f, orders_cv, 'r-');
title('Adaptive Parameters (Window & Order)');
xlabel('Freq'); ylabel('Value'); legend('Window', 'Order'); grid on;

% Save to the newly specified file name
filtered_matrix = [f, mag_cv];
writematrix(filtered_matrix, 'best_filtered4.csv');
fprintf('Filtered data successfully saved to "best_filtered4.csv"\n');

disp('All done.');

function [mag_smooth, opt_windows, opt_orders] = adaptive_sg_cv(f, mag, W_cands, P_cands)
    N = length(mag);
    f = f(:); mag = mag(:);
    W_cands = W_cands + ~mod(W_cands, 2); % Ensure windows are odd
    W_cands = unique(W_cands);
    opt_windows = zeros(N,1);
    opt_orders = zeros(N,1);
    mag_smooth = zeros(N,1);
    h = waitbar(0, 'CV optimization running...');
    
    for i = 1:N
        best_aic = inf;
        best_w = W_cands(1);
        best_p = P_cands(1);
        best_intercept = mag(i); % Fallback initialization
        
        for w = W_cands
            half = floor(w/2);
            idx = max(1, i-half):min(N, i+half);
            L = length(idx);
            
            % Local coordinate shifting to prevent numerical issues
            x = f(idx) - f(i);
            % ... [Function logic continues based on standard SG implementation]
        end
    end
end
